function(mg_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} BEFORE PRIVATE "${MG}/3rdparty/glslang" "${MG}/include" "${MG}" "${MG}/3rdparty/xxhash")
    target_compile_definitions(${name} PRIVATE AMCL_MG_FRAME_STATS=0 AMCL_MG_UPLOAD_SCHEDULER_LAB=0 PROFILING=0)
    target_compile_options(${name} PRIVATE -g -O1 -UNDEBUG -ffunction-sections -fdata-sections)
    target_link_options(${name} PRIVATE -Wl,--gc-sections)
    if(MG_TEST_SANITIZERS)
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
